cmake_minimum_required(VERSION 3.24)
project(HiveDesk VERSION 0.1.0 LANGUAGES CXX)
set(HIVEDESK_VERSION "0.1.0-alpha.34")
add_compile_definitions(HIVEDESK_VERSION="${HIVEDESK_VERSION}")
enable_language(RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "HiveDesk requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/EHsc> $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
file(READ "${CMAKE_SOURCE_DIR}/resources/update-public-key.json" HIVEDESK_UPDATE_KEY)
string(JSON HIVEDESK_UPDATE_MODULUS GET "${HIVEDESK_UPDATE_KEY}" modulus)
string(JSON HIVEDESK_UPDATE_EXPONENT GET "${HIVEDESK_UPDATE_KEY}" exponent)
configure_file(src/update_key.h.in generated/update_key.h @ONLY)
add_library(hive_update STATIC src/update.cpp)
target_include_directories(hive_update PUBLIC include PRIVATE "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(hive_update PUBLIC bcrypt crypt32 advapi32 shell32 ole32 winhttp version)
add_executable(HiveDeskUpdater src/updater.cpp)
target_link_libraries(HiveDeskUpdater PRIVATE hive_update userenv)
add_executable(HiveDeskGpuPolicy src/gpu_policy_main.cpp src/gpu_policy.cpp)
target_include_directories(HiveDeskGpuPolicy PRIVATE src third_party/nvapi)
target_link_libraries(HiveDeskGpuPolicy PRIVATE shell32 ole32 advapi32)
add_library(hive_core STATIC src/transport.cpp src/media.cpp src/media_cuda.cpp src/config.cpp src/desktop_aux.cpp src/adaptive.cpp src/input.cpp src/input_capture.cpp src/host_bridge.cpp src/telemetry.cpp)
target_include_directories(hive_core PUBLIC include third_party/nvcodec/include)
target_link_libraries(hive_core PUBLIC hive_update ws2_32 iphlpapi advapi32 bcrypt crypt32 d3d11 dxgi d3dcompiler user32 gdi32 ole32 oleaut32 windowscodecs dwmapi)
add_executable(HiveDesk WIN32 src/main.cpp src/shell.cpp tests/host_check.cpp resources/app.rc resources/app.manifest)
target_include_directories(HiveDesk PRIVATE resources)
target_link_libraries(HiveDesk PRIVATE hive_core shell32 ole32 dcomp d2d1 msimg32)
add_executable(HiveDeskService src/service.cpp)
target_link_libraries(HiveDeskService PRIVATE hive_update advapi32 userenv wtsapi32)
add_executable(HiveDeskHost src/host.cpp)
target_link_libraries(HiveDeskHost PRIVATE hive_core wtsapi32)
add_executable(HiveDeskCheck tests/check.cpp)
target_link_libraries(HiveDeskCheck PRIVATE hive_core)
enable_testing()
add_executable(HiveDeskGpuPolicyCheck tests/gpu_policy.cpp src/gpu_policy.cpp)
target_include_directories(HiveDeskGpuPolicyCheck PRIVATE src)
add_test(NAME gpu_policy COMMAND HiveDeskGpuPolicyCheck)
set_tests_properties(gpu_policy PROPERTIES TIMEOUT 15)
add_executable(HiveDeskGpuPolicyRestartCheck tests/gpu_policy_restart.cpp)
target_link_libraries(HiveDeskGpuPolicyRestartCheck PRIVATE hive_update)
add_test(NAME gpu_policy_restart COMMAND HiveDeskGpuPolicyRestartCheck)
set_tests_properties(gpu_policy_restart PROPERTIES TIMEOUT 20)

target_include_directories(HiveDeskGpuPolicyDriverCheck PRIVATE src third_party/nvapi)
target_link_libraries(HiveDeskGpuPolicyDriverCheck PRIVATE shell32 ole32 advapi32)
